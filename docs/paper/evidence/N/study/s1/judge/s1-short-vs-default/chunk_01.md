```python
def add_to_sequence(sequence):
    if sequence == 0:
        return 0
    else:
        for i in range(1, len(sequence)):
            if sequence[i] == sequence[i + 1]:
                return sequence[i + 1]
            else:
                return 0
        else:
            return 0
    return sequence

# Example usage:

sequence = add_to_sequence(5)
print(sequence)  # Output: 4
```
